/*PROGRAM OF INSERTION*/

#include<conio.h>
#include<stdio.h>
void main()
{
int i,a[15],b,k,x;
clrscr();
printf("Enter the no.of element:-");
scanf("%d",&b);
printf("Enter the element:-\n");
for(i=0;i<b;i++)
scanf("%d",&a[i]);
printf("Enter the position where element to be inserted:- ");
scanf("%d",&k);
for(i=b-1;i>=k-1;i--)
{
a[i+1]=a[i];
}
printf("Enter the element to be inserted:- ");
scanf("%d",&x);
a[k-1]=x;
b=b+1;
printf("The elements after insertion:- ");
for(i=0;i<b;i++)
printf("%d ",a[i]);
getch();
}
