/* Selection Sort*/


#include<stdio.h>
#include<conio.h>
void main()
{
 int n,j,i,c,a[10],min,pos;
 clrscr();
 printf("Enter the size of the Array(<10):");
 scanf("%d",&n);
 printf("\nEnter the elements in the Array:--\n");
 for(i=0;i<n;i++)
  scanf("%d",&a[i]);
  min=a[0];
 for(i=0;i<n;i++)
  {
   min=a[i];
   pos=i;
   for(j=i+1;j<=n-1;j++)
    {
     if(min>a[j])
      {
       pos=j;
       min=a[j];
      }
    }
   c=a[i];
   a[i]=a[pos];
   a[pos]=c;
  }
  printf("\nThe Sorted Array(Ascending) in as follows:--\n");
  for(i=0;i<n;i++)
   printf("%d ",a[i]);
 getch();
}
