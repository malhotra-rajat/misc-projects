
/*PROGRAM OF BUBBLE SORT*/

{

for(i=0;i<b;i++)
scanf("%d",&a[i]);
for(i=0;i<b;i++)
{for(j=0;j<b-i;j++)
{
if(a[j]<a[j+1])
{temp=a[j];

}}}

{printf("%d ",a[i]);
}

}
