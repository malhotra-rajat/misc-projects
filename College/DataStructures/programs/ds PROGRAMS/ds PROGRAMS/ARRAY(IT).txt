void traverse(int a[],int y)
{

}

{
int i,k,z,item;
printf("\nEnter the position where element to be inserted:");
scanf("%d",&k);
printf("\nEnter the element:");
scanf("%d",&item);
for(z=y-1;z>=k;z--)
a[z+1]=a[z];
a[k]=item;
y=y+1;
printf("\nNew list is:");
for(i=0;i<y;i++)
printf("%d,",a[i]);
return;
}
